"""Report the common Direct/DF runner without certifying unmatched branches."""
import hashlib
import json
from pathlib import Path

root = Path('.artifacts/issue385/direct-protocol')
load = lambda p: json.loads(p.read_text())
rows = []
for path in sorted(root.rglob('*.json')):
    if path.name in ('run.json', 'summary.json'):
        continue
    data = load(path)
    assert data['benchmark'] == 'compare_gpu4pyscf_batch'
    row = {'artifact': str(path.relative_to(root)),
           'sha256': hashlib.sha256(path.read_bytes()).hexdigest(),
           'aos': data['workload']['ao_count'], 'batch': data['workload']['batch_size'],
           'gate': data['gate'], 'all_repeat_accuracy': data['accuracy'],
           'iteration_matched': data['timing_summary']['iteration_matched']}
    for engine in ('vibeqc', 'gpu4pyscf'):
        samples = data[engine]['warm_samples']
        assert len(samples) == 5
        row[engine] = {'seconds': [s['seconds'] for s in samples],
                       'median_seconds': data[engine]['warm_median_seconds'],
                       'scf_branches': [[c['iterations'] for c in s['convergence']] for s in samples]}
    rows.append(row)

for aos in (384, 768):
    a,b = [load(root / f'{aos}-{variant}.json') for variant in ('baseline','candidate')]
    assert a['workload'] == b['workload']
    for data in (a,b):
        assert data['gate']['passed']
        assert data['accuracy']['maximum_energy_error_hartree'] <= 1e-9
        assert data['accuracy']['maximum_force_error_hartree_per_bohr'] <= 1e-8
        assert all(s['convergence'][0]['iterations'] == 3 for s in data['vibeqc']['warm_samples'])


def flatten(value):
    if isinstance(value, list):
        return [x for child in value for x in flatten(child)]
    return [value]


small_parity = []
for batch in (1,4):
    filename = f'water-tetramer-def2-svp-spherical-b{batch}-cuda.json'
    a,b = [load(root / folder / filename) for folder in ('matrix-96-baseline','matrix-96')]
    assert a['workload'] == b['workload']
    row = {'aos':96, 'batch':batch, 'baseline_gate':a['gate'], 'candidate_gate':b['gate']}
    for key in ('energies_hartree','forces_hartree_per_bohr'):
        errors = [max(abs(x-y) for x,y in zip(flatten(s[key]), flatten(t[key]), strict=True)) for s,t in zip(a['vibeqc']['warm_samples'],b['vibeqc']['warm_samples'],strict=True)]
        assert len(errors) == 5 and max(errors) <= 3e-11
        row['native_baseline_candidate_maximum_' + key + '_difference'] = max(errors)
    small_parity.append(row)

payload = {'protocol': 'Unmodified compare_gpu4pyscf_batch.py, shared with Direct J/K; real_molecule_gate.py small DF points retain original criteria.',
           'all_clean_repeats_per_engine':5,
           'interpretation': 'Same measurement procedure; raw unmatched SCF branches remain explicitly unmatched. Do not use ordinary cross-engine ratios as an iteration-matched headline. Large native baseline/candidate settings and branches are equal.',
           'large_native_before_after': {str(aos): {variant: load(root / f'{aos}-{variant}.json')['vibeqc']['warm_median_seconds'] for variant in ('baseline','candidate')} for aos in (384,768)},
           'matrix_96_native_parity':small_parity, 'rows':rows}
(root / 'summary.json').write_text(json.dumps(payload,indent=2)+'\n')
print('Common Direct/DF protocol summarized; original gates and all five samples retained.')
