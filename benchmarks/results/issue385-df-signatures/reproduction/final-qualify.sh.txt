#!/usr/bin/env bash
set -euo pipefail
export PYTHONPATH=python:. OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1
export PATH="/group/software/cuda-12.9.1/bin:$PATH"
# Compilation owns the allocation while the GPU is idle, so local timing jobs
# cannot overlap this resource-intensive build.
cmake --build --preset cuda-release-sm120 --target vibeqc vibeqc_df_occupied_response_tests vibeqc_df_shell_pairs_tests --parallel 6 > .artifacts/issue385/final-build.log 2>&1
mkdir -p .artifacts/issue385/final
cp build/cuda-release-sm120/libvibeqc.so .artifacts/issue385/final/libvibeqc.so
ln -s libvibeqc.so .artifacts/issue385/final/libvibeqc.so.0
cp build/cuda-release-sm120/vibeqc_df_shell_pairs_tests build/cuda-release-sm120/vibeqc_df_occupied_response_tests .artifacts/issue385/final/
git diff HEAD --binary -- src python tests > .artifacts/issue385/final/source.patch
export VIBEQC_LIBRARY="$PWD/.artifacts/issue385/final/libvibeqc.so"
export LD_LIBRARY_PATH="$PWD/.artifacts/issue385/final:/group/software/cuda-12.9.1/lib64"
export VIBEQC_RESOURCE_CUDA_TEST=1
.artifacts/issue385/final/vibeqc_df_shell_pairs_tests
.artifacts/issue385/final/vibeqc_df_occupied_response_tests
/tmp/vibeqc-pr-review-env/bin/python -m pytest -q tests/python/test_df_shell_derivatives_cuda.py tests/python/test_df_occupied_response_cuda.py tests/python/test_df_resident_response_cuda.py > .artifacts/issue385/final/tests.log 2>&1

# CPU contracts use the same frozen library; profile selection stays disabled.
VIBEQC_PROFILE=off /tmp/vibeqc-pr-review-env/bin/python -m pytest -q tests/python/test_hf_resources.py tests/python/test_df_resources.py tests/python/test_benchmarks.py tests/python/test_compiler_structure.py tests/python/test_scf_structure.py tests/python/test_cuda_ownership.py tests/python/test_hf_resources_cuda.py > .artifacts/issue385/final/contracts.log 2>&1
compute-sanitizer --tool memcheck --error-exitcode 99 .artifacts/issue385/final/vibeqc_df_shell_pairs_tests > .artifacts/issue385/final/memcheck-shell.log 2>&1
compute-sanitizer --tool synccheck --error-exitcode 99 .artifacts/issue385/final/vibeqc_df_shell_pairs_tests > .artifacts/issue385/final/synccheck-shell.log 2>&1
compute-sanitizer --tool memcheck --error-exitcode 99 .artifacts/issue385/final/vibeqc_df_occupied_response_tests > .artifacts/issue385/final/memcheck-response.log 2>&1
compute-sanitizer --tool memcheck --error-exitcode 99 /tmp/vibeqc-pr-review-env/bin/python -m pytest -q tests/python/test_df_occupied_response_cuda.py::test_packed_response_crosses_ao_blocks_and_auxiliary_panels > .artifacts/issue385/final/memcheck-ragged.log 2>&1
