// Replay Nsight launch resources through CUDA's standalone occupancy calculator.
// This is a theoretical resident-block limit, never achieved occupancy/stalls.
#include <cuda_runtime_api.h>
#include <cuda_occupancy.h>
#include <cstdlib>
#include <iostream>
int main() {
  if (!std::getenv("SLURM_JOB_ID")) return 2;
  cudaDeviceProp device{};
  if (cudaGetDeviceProperties(&device, 0) != cudaSuccess) return 3;
  cudaOccDeviceProp properties(device);
  cudaOccDeviceState state;
  std::cout << "{\"device\":\"" << device.name << "\",\"sm_threads\":"
            << device.maxThreadsPerMultiProcessor << ",\"reserved_shared_per_block\":"
            << device.reservedSharedMemPerBlock << ",\"rows\":[";
  int registers, block; std::size_t shared, dynamic; bool first=true;
  while (std::cin >> registers >> shared >> dynamic >> block) {
    cudaOccFuncAttributes attributes;
    attributes.maxThreadsPerBlock=device.maxThreadsPerBlock;
    attributes.numRegs=registers; attributes.sharedSizeBytes=shared;
    cudaOccResult result{};
    if (cudaOccMaxActiveBlocksPerMultiprocessor(&result, &properties, &attributes,
                                               &state, block, dynamic) != CUDA_OCC_SUCCESS) return 4;
    if (!first) std::cout << ',';
    first=false;
    std::cout << "{\"registers\":" << registers << ",\"static_shared\":" << shared
              << ",\"dynamic_shared\":" << dynamic << ",\"block_threads\":" << block
              << ",\"resident_blocks\":" << result.activeBlocksPerMultiprocessor
              << ",\"theoretical_occupancy\":"
              << double(result.activeBlocksPerMultiprocessor*block)/device.maxThreadsPerMultiProcessor
              << ",\"limiting_factors\":" << result.limitingFactors << '}';
  }
  std::cout << "]}\n";
}
