"""Audit the installed reference's actual DF model separately from all timings."""
import hashlib
import inspect
import json
import os
from pathlib import Path

import cupy as cp
import numpy as np
from gpu4pyscf.df import df as gpu_df
from gpu4pyscf.df import int3c2e_bdiv
from pyscf import gto

from benchmarks._cases import benchmark_cases
from benchmarks.df_policy_endpoint import CASES

assert os.environ.get('SLURM_JOB_ID')
rows = []
for aos in (384, 768):
    case = benchmark_cases()[CASES[aos]]
    native_path = Path('.artifacts/issue385/packet') / f'{aos}-clean.json'
    native = json.loads(native_path.read_text())
    mol = gto.M(atom=case.atoms, basis=case.pyscf_basis, unit='Bohr', cart=False, verbose=0)
    np.testing.assert_array_equal(mol.atom_coords(), np.array([a[1] for a in case.atoms]))
    provider = gpu_df.DF(mol, auxbasis=case.pyscf_basis)
    branches = []
    original = gpu_df._decompose_j2c

    def observed(*args, **kwargs):
        result = original(*args, **kwargs)
        branches.append(result[1])
        return result

    gpu_df._decompose_j2c = observed
    try:
        provider.build()
        cp.cuda.Stream.null.synchronize()
    finally:
        gpu_df._decompose_j2c = original
    metric = provider.auxmol.intor('int2c2e')
    values = np.linalg.eigvalsh(metric)
    native_metric = native['samples'][0]['metric'][0]
    cutoff = native_metric['absolute_threshold']
    rank = int(np.count_nonzero(values > cutoff))
    assert rank == native_metric['effective_rank'] == provider.naux == aos
    assert branches and set(branches) == {'cd'}
    np.testing.assert_allclose(values[-1] / values[0], native_metric['condition_number'], rtol=1e-9)
    rows.append({
        'aos': aos, 'slurm_job_id': os.environ['SLURM_JOB_ID'],
        'native_record_sha256': hashlib.sha256(native_path.read_bytes()).hexdigest(),
        'same': ['geometry', 'neutral singlet RHF', 'spherical def2-SVP orbital and auxiliary bases',
                 'full public auxiliary rank', 'FP64 DF tensors and metric', 'complete analytic atom forces'],
        'differences': {
            'native_scf_stopping': 'energy change 1e-12 and density RMS 1e-10',
            'gpu4_scf_stopping': 'conv_tol 1e-12 and orbital gradient conv_tol_grad 1e-10',
            'native_screening_control': native['scientific_settings']['screening_tolerance'],
            'gpu4_df_optimizer_cutoff': inspect.signature(int3c2e_bdiv.Int3c2eOpt.build).parameters['cutoff'].default,
            'native_metric': 'eigensystem with relative cutoff 1e-10',
            'gpu4_metric': 'observed successful Cholesky, no eigentruncation',
            'gpu4_fallback_absolute_threshold_unused': float(gpu_df.LINEAR_DEP_THR),
            'warm_density': 'Each engine freezes its own post-cold converged density; they are not byte-identical shared seeds.',
            'native_force_timing': 'Five host-traced force-stage intervals inside native three-iteration endpoints',
            'gpu4_force_timing': 'Five clean repeated complete gradients from one converged state; both edges synchronized, host result included',
        },
        'metric_check': {'public_rank': rank, 'minimum_eigenvalue': float(values[0]),
                         'maximum_eigenvalue': float(values[-1]), 'condition_number': float(values[-1]/values[0]),
                         'native_absolute_threshold': cutoff, 'observed_reference_decompositions': branches,
                         'metric_dtype': str(metric.dtype)},
        'reference_sources': {name: hashlib.sha256(Path(inspect.getfile(module)).read_bytes()).hexdigest()
                              for name, module in [('gpu4_df', gpu_df), ('gpu4_int3c', int3c2e_bdiv)]},
        'interpretation': 'Same full-rank DF model and validated observable, with explicitly different algorithms and stopping/screening policies. No fully iteration-matched cross-engine endpoint claim.'})
    provider.reset()
    cp.get_default_memory_pool().free_all_blocks()
Path('.artifacts/issue385/reference/conditions.json').write_text(json.dumps(rows, indent=2) + '\n')
print('Reference model/metric audit passed; implementation and timing differences retained.')
