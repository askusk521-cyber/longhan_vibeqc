#!/usr/bin/env bash
set -euo pipefail
export PYTHONPATH=python:. OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1
export PATH="/group/software/cuda-12.9.1/bin:$PATH"
export VIBEQC_LIBRARY="$PWD/.artifacts/issue385/final/libvibeqc.so"
export LD_LIBRARY_PATH="$PWD/.artifacts/issue385/final:/group/software/cuda-12.9.1/lib64"
nvidia-smi --query-gpu=name,driver_version,memory.total,pstate,clocks.sm,clocks.mem,power.draw,temperature.gpu --format=csv > .artifacts/issue385/final/hardware-before.csv
for aos in 96 192 384 768; do
  case "$aos" in
    96) case_name=water-tetramer-def2-svp-spherical ;;
    192) case_name=water-octamer-s4-def2-svp-spherical ;;
    384) case_name=water-hexadecamer-2s4-def2-svp-spherical ;;
    768) case_name=water-32mer-4s4-def2-svp-spherical ;;
  esac
  reference="benchmarks/results/issue377-379-df/gpu4pyscf/$case_name.json"
  /tmp/vibeqc-pr-review-env/bin/python -m benchmarks.df_policy_endpoint --aos "$aos" --control VIBEQC_DF_PRIMITIVE_BUCKETS --policies off auto --repeats 5 --reference "$reference" --output ".artifacts/issue385/final/$aos-clean.json"
  nvidia-smi --query-compute-apps=pid,used_memory --format=csv,noheader,nounits -lms 100 > ".artifacts/issue385/final/$aos-memory.csv" &
  sampler_pid=$!
  trap 'kill "$sampler_pid" 2>/dev/null || true' EXIT
  VIBEQC_DF_SHELL_COUNTERS=1 /tmp/vibeqc-pr-review-env/bin/python -m benchmarks.df_policy_endpoint --aos "$aos" --control VIBEQC_DF_PRIMITIVE_BUCKETS --policies off auto --repeats 1 --trace --reference "$reference" --output ".artifacts/issue385/final/$aos-work.json"
  kill "$sampler_pid"
  trap - EXIT
  if [[ "$aos" == 384 || "$aos" == 768 ]]; then
    /tmp/vibeqc-pr-review-env/bin/python -m benchmarks.df_policy_endpoint --aos "$aos" --control VIBEQC_DF_PRIMITIVE_BUCKETS --policies auto --repeats 5 --host-trace --reference "$reference" --output ".artifacts/issue385/final/$aos-force-stage.json"
  fi
done
/tmp/vibeqc-pr-review-env/bin/python .artifacts/issue385/check_final.py
nvidia-smi --query-gpu=name,driver_version,memory.total,pstate,clocks.sm,clocks.mem,power.draw,temperature.gpu --format=csv > .artifacts/issue385/final/hardware-after.csv
# Refresh the exact #383 binary separately, preserving its frozen source patch.
export VIBEQC_LIBRARY=/home/jzzeng/codes/vibeqc-issue382/.artifacts/issue385/baseline/libvibeqc.so
export LD_LIBRARY_PATH=/home/jzzeng/codes/vibeqc-issue382/.artifacts/issue385/baseline:/group/software/cuda-12.9.1/lib64
mkdir -p .artifacts/issue385/baseline-refresh
for aos in 384 768; do
  case "$aos" in
    384) case_name=water-hexadecamer-2s4-def2-svp-spherical ;;
    768) case_name=water-32mer-4s4-def2-svp-spherical ;;
  esac
  /tmp/vibeqc-pr-review-env/bin/python -m benchmarks.df_policy_endpoint --aos "$aos" --control VIBEQC_DF_DERIVATIVE_PAIRS --policies auto --repeats 5 --reference "benchmarks/results/issue377-379-df/gpu4pyscf/$case_name.json" --output ".artifacts/issue385/baseline-refresh/$aos-clean.json"
done
# Audit actual metric decomposition and settings separately from timed work.
LD_LIBRARY_PATH=/tmp/vibeqc-308-gpu4pyscf-env/lib/python3.13/site-packages/cutensor/lib:/group/software/cuda-12.9.1/lib64 /tmp/vibeqc-308-gpu4pyscf-env/bin/python .artifacts/issue385/reference/audit_conditions.py
# Keep the independent Direct-vs-Direct gate and its stricter original tolerances.
export VIBEQC_LIBRARY="$PWD/.artifacts/issue385/final/libvibeqc.so"
export LD_LIBRARY_PATH="$PWD/.artifacts/issue385/final:/tmp/vibeqc-308-gpu4pyscf-env/lib/python3.13/site-packages/cutensor/lib:/group/software/cuda-12.9.1/lib64"
/tmp/vibeqc-308-gpu4pyscf-env/bin/python benchmarks/real_molecule_gate.py --density-fitting none --repeats 5 --output-directory .artifacts/issue385/direct > .artifacts/issue385/direct-gate.log 2>&1
