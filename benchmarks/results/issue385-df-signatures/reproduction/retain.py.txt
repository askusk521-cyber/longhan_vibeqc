"""Retain reviewed issue 385 JSON losslessly; keep binaries and timelines local."""
import hashlib
import json
import shutil
from pathlib import Path

SOURCE = Path('.artifacts/issue385')
ROOT = Path('benchmarks/results/issue385-df-signatures')
ROOT.mkdir(parents=True, exist_ok=True)


def digest(data):
    return hashlib.sha256(data).hexdigest()


def pretty(value, level=0):
    """Keep scalar vectors compact while retaining readable sample objects."""
    if isinstance(value, dict):
        if not value:
            return '{}'
        return '{\n' + ',\n'.join('  '*(level+1) + json.dumps(k) + ': ' + pretty(v, level+1)
                                 for k, v in value.items()) + '\n' + '  '*level + '}'
    if isinstance(value, list) and any(isinstance(v, (list, dict)) for v in value):
        return '[\n' + ',\n'.join('  '*(level+1) + pretty(v, level+1) for v in value) + '\n' + '  '*level + ']'
    return json.dumps(value, allow_nan=False)


manifest = {
    'base_commit': '422796f04553953a65e45bf1f0e41642f482a591',
    'reduction': 'Only repeated basis_metadata objects are interned by content hash; expand basis_metadata_ref using the per-file catalog to recover original parsed JSON. Raw timings, forces, gates and source identities are unchanged.',
    'files': [],
}
for folder in ('baseline', 'baseline-refresh', 'uniform', 'packet', 'final', 'reference', 'direct', 'direct-baseline', 'direct-tighter-reference', 'direct-protocol'):
    for source in sorted((SOURCE / folder).rglob('*.json')):
        original = json.loads(source.read_bytes())
        catalog = {}

        def reduce(value):
            if isinstance(value, list):
                return [reduce(v) for v in value]
            if not isinstance(value, dict):
                return value
            result = {}
            for k, v in value.items():
                if k == 'basis_metadata':
                    key = digest(json.dumps(v, sort_keys=True).encode())
                    catalog[key] = v
                    result['basis_metadata_ref'] = key
                else:
                    result[k] = reduce(v)
            return result

        def expand(value):
            if isinstance(value, list):
                return [expand(v) for v in value]
            if isinstance(value, dict):
                return {'basis_metadata' if k == 'basis_metadata_ref' else k:
                        catalog[v] if k == 'basis_metadata_ref' else expand(v)
                        for k, v in value.items()}
            return value

        reduced = reduce(original)
        assert expand(reduced) == original
        if catalog:
            reduced['basis_metadata_catalog'] = catalog
        target = ROOT / source.relative_to(SOURCE)
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_text(pretty(reduced) + '\n')
        assert target.stat().st_size <= 1024**2, target
        manifest['files'].append({'path': str(target.relative_to(ROOT)),
                                  'original_sha256': digest(source.read_bytes()),
                                  'retained_sha256': digest(target.read_bytes()),
                                  'lossless_parsed_json_roundtrip': True})
for name in ('histogram.json', 'baseline-uniform-occupancy.json', 'qualification.json', 'rejection-history.json', 'direct-audit.json'):
    source = SOURCE / name
    if source.exists():
        shutil.copyfile(source, ROOT / name)
reproduction = ROOT / 'reproduction'
reproduction.mkdir(exist_ok=True)
for folder in ('uniform', 'packet', 'final'):
    source = SOURCE / folder / 'source.patch'
    if source.exists():
        shutil.copyfile(source, reproduction / f'{folder}-source.patch')
# Preserve measured runner bytes despite formatter evolution. Python can execute
# these historical sources directly; they are not transient command output.
for name in ('histogram.py', 'profile_endpoint.py', 'reduce_nsys.py', 'report_components.py',
             'occupancy.cpp', 'packet-run.sh', 'final-run.sh', 'check_final.py', 'qualify_evidence.py', 'final-sanitize.sh', 'final-qualify.sh', 'audit_direct_results.py', 'direct-audit.sh', 'direct_protocol.py', 'summarize_direct_protocol.py', 'retain.py', 'reference/gpu4_reference.py', 'reference/audit_conditions.py'):
    source = SOURCE / name
    if source.exists():
        target = reproduction / (source.name + '.txt')
        shutil.copyfile(source, target)
        manifest['files'].append({'path': str(target.relative_to(ROOT)), 'retained_sha256': digest(target.read_bytes()), 'role': 'measured or reproduction source'})
shutil.copyfile(Path('benchmarks/df_policy_endpoint.py'), reproduction / 'df_policy_endpoint.py.txt')
# Include hashes for ancillary tables, reconstruction patches and the README
# as well as the original/retained pairs above. Omit the self-referential manifest.
indexed = {entry['path'] for entry in manifest['files']}
for target in sorted(ROOT.rglob('*')):
    if target.is_file() and target.name != 'manifest.json':
        relative = str(target.relative_to(ROOT))
        assert target.stat().st_size <= 1024**2, target
        if relative not in indexed:
            manifest['files'].append({'path': relative, 'retained_sha256': digest(target.read_bytes()), 'role': 'reviewed ancillary evidence or source'})
(ROOT / 'manifest.json').write_text(pretty(manifest) + '\n')
print('Retained', len(manifest['files']), 'records/sources; all parsed JSON roundtrips passed.')
