"""Assert automatic routing, unchanged semantic work, and retained small gates."""
import json
from pathlib import Path

root = Path('.artifacts/issue385/final')
histograms = {h['aos']: h for h in json.loads(Path('.artifacts/issue385/histogram.json').read_text())}
for aos in (96, 192, 384, 768):
    clean = json.loads((root / f'{aos}-clean.json').read_text())
    assert len(clean['samples']) == 10
    for sample in clean['samples']:
        energy_gate, force_gate = {96: (3e-11, 3e-11), 192: (1e-10, 5e-10)}.get(aos, (1e-9, 1e-8))
        assert sample['maximum_energy_error'] <= energy_gate
        assert sample['maximum_force_error'] <= force_gate
    work = json.loads((root / f'{aos}-work.json').read_text())
    counters = {s['policy']: next(g for g in s['components']['groups'] if g['operation'] == 'force_response')['counter_sums'] for s in work['samples']}
    off, auto = counters['off'], counters['auto']
    assert auto['shell_primitive_signature_policy'] == (2 if aos in (384, 768) else 0)
    assert off['shell_primitive_signature_policy'] == 0
    for name in ('shell_triples_visited', 'shell_triples_nonzero', 'shell_primitive_products',
                 'shell_cartesian_component_products', 'shell_public_weights_nonzero',
                 'shell_public_weights_consumed', 'three_center_derivative_weights',
                 'metric_derivative_weights', 'device_to_host_bytes', 'stream_synchronizations'):
        assert off.get(name) == auto.get(name), (aos, name, off.get(name), auto.get(name))
    if aos in histograms:
        h = histograms[aos]
        assert auto['shell_triples_visited'] == h['tasks']
        assert auto['shell_primitive_products'] == h['primitive_products']
        assert auto['three_center_signature_packet_launches'] == h['launches']
        assert auto['host_to_device_bytes'] - off['host_to_device_bytes'] == auto['shell_signature_id_bytes']
print('Final automatic routing, semantic work and all retained numerical gates passed.')
