"""Reconstruct executed shell grids; bounded host chunks model warp trip-count padding.

This is a loop-count model, not a measured stall fraction. Independent native
visited/nonzero/primitive counters and Nsight grid sizes must validate the domain.
"""
import argparse
import hashlib
import json
import math
import re
from collections import Counter
from pathlib import Path

import numpy as np
from vibeqc import Atom
from vibeqc.calculator import _named_basis_shells
from benchmarks._cases import benchmark_cases
from benchmarks.df_policy_endpoint import CASES


def analyze(aos):
    case = benchmark_cases()[CASES[aos]]
    shells = _named_basis_shells(case.vibeqc_basis, [Atom.from_value(a) for a in case.atoms])
    angular = np.array([s.angular_momentum for s in shells])
    nprim = np.array([len(s.primitives) for s in shells])
    offsets = np.r_[0, np.cumsum(2 * angular + 1)]
    assert offsets[-1] == aos
    # Exact #383 retained routes: dense 52-AO panels and packed aligned <=64.
    panels = []
    start = 0
    while start < aos:
        end = min(aos, start + (52 if aos == 384 else 64))
        if aos == 768:
            end = int(offsets[np.searchsorted(offsets, end, side='right') - 1])
        panels.append((start, end))
        start = end
    generated = Path('build/cuda-release-sm120/generated/generated_df_shell_derivatives.cuh').read_text()
    schedules = {tuple(map(int, m[:3])): tuple(map(int, m[3:])) for m in re.findall(
        r'Schedule<(\d),(\d),(\d),2>\s*\{\s*static constexpr unsigned lanes=(\d+);\s*static constexpr unsigned groups=(\d+);', generated)}
    rows, signature = {}, Counter()
    for lo in range(4):
        ia = np.flatnonzero(angular == lo)
        for lb in range(lo + 1):
            ib = np.flatnonzero(angular == lb)
            if not len(ia) or not len(ib):
                continue
            if lo == lb:
                ai, bi = np.tril_indices(len(ia))
                ap, bp = nprim[ia[ai]], nprim[ib[bi]]
            else:
                ap = np.repeat(nprim[ia], len(ib))
                bp = np.tile(nprim[ib], len(ia))
            pair_signatures = Counter(zip(ap.tolist(), bp.tolist()))
            for lc in range(4):
                key = (lo, lb, lc)
                lanes, groups = schedules[key]
                width = 32 // lanes
                row = {'angular': key, 'lanes': lanes, 'groups': groups, 'launches': 0,
                       'tasks': 0, 'primitive_products': 0, 'warp_padded_primitive_slots': 0, 'block_padded_primitive_slots': 0,
                       'grid_blocks': [], 'panel_shells': [], 'signatures': []}
                for start, end in panels:
                    ci = np.flatnonzero((angular == lc) & (offsets[:-1] < end) & (offsets[1:] > start))
                    if not len(ci):
                        continue
                    cp = nprim[ci]
                    tasks = len(ap) * len(cp)
                    row['launches'] += 1
                    row['tasks'] += tasks
                    row['grid_blocks'].append(math.ceil(tasks / groups))
                    row['panel_shells'].append(len(ci))
                    row['primitive_products'] += int(np.sum(ap * bp)) * int(np.sum(cp))
                    for (na, nb), nab in pair_signatures.items():
                        for nc, ncc in Counter(cp.tolist()).items():
                            signature[key + (na, nb, nc)] += nab * ncc
                    # Physical-warp order follows native auxiliary-fast task IDs.
                    # Chunks are warp-aligned and never hold a resident N^3 list.
                    for begin in range(0, tasks, 65536):
                        indices = np.arange(begin, min(begin + 65536, tasks))
                        products = ap[indices // len(cp)] * bp[indices // len(cp)] * cp[indices % len(cp)]
                        warp_products = np.pad(products, (0, (-len(products)) % width))
                        row['warp_padded_primitive_slots'] += int(warp_products.reshape(-1, width).max(axis=1).sum()) * width
                        block_products = np.pad(products, (0, (-len(products)) % groups))
                        row['block_padded_primitive_slots'] += int(block_products.reshape(-1, groups).max(axis=1).sum()) * groups
                if row['tasks']:
                    rows[key] = row
    for key, tasks in sorted(signature.items()):
        np3 = key[3:]
        rows[key[:3]]['signatures'].append({'nprim': np3, 'tasks': tasks, 'primitive_products': tasks * math.prod(np3)})
    for row in rows.values():
        p = row['primitive_products']
        row['modeled_warp_efficiency'] = p / row['warp_padded_primitive_slots']
        row['modeled_block_efficiency'] = p / row['block_padded_primitive_slots']
        row['task_primitive_min'] = min(math.prod(s['nprim']) for s in row['signatures'])
        row['task_primitive_max'] = max(math.prod(s['nprim']) for s in row['signatures'])
        mean = p / row['tasks']
        variance = sum(s['tasks'] * (math.prod(s['nprim']) - mean)**2 for s in row['signatures']) / row['tasks']
        row['task_primitive_cv'] = math.sqrt(variance) / mean
    total_tasks = sum(r['tasks'] for r in rows.values())
    total_products = sum(r['primitive_products'] for r in rows.values())
    # Independent actual device counters, retained from the #383 implementation.
    retained = json.loads(Path(f'benchmarks/results/issue382-packed-df/final/{aos}-work.json').read_text())
    counters = next(g['counter_sums'] for g in retained['samples'][0]['components']['groups'] if g['operation'] == 'force_response')
    assert total_tasks == counters['shell_triples_visited'] == counters['shell_triples_nonzero']
    assert total_products == counters['shell_primitive_products']
    return {'aos': aos, 'shells': len(shells), 'panels': panels,
            'shell_histogram': [{'angular': l, 'nprim': p, 'shells': n} for (l,p),n in sorted(Counter(zip(angular.tolist(),nprim.tolist())).items())],
            'tasks': total_tasks, 'primitive_products': total_products,
            'modeled_warp_efficiency': total_products / sum(r['warp_padded_primitive_slots'] for r in rows.values()),
            'modeled_block_efficiency': total_products / sum(r['block_padded_primitive_slots'] for r in rows.values()),
            'launches': sum(r['launches'] for r in rows.values()),
            'generated_schedule_sha256': hashlib.sha256(generated.encode()).hexdigest(),
            'classes': list(rows.values()),
            'interpretation': 'Exact shell-domain counts validated against independent native device counters. Warp padding models primitive product trip counts only; it is not measured elapsed-time or a predicted speedup. Every triple is active in these retained workloads.'}


if __name__ == '__main__':
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--output', type=Path, required=True)
    args = parser.parse_args()
    result = [analyze(n) for n in (384,768)]
    args.output.write_text(json.dumps(result, indent=2) + '\n')
    for r in result:
        print(r['aos'],r['tasks'],r['primitive_products'],r['modeled_warp_efficiency'],r['shell_histogram'])
