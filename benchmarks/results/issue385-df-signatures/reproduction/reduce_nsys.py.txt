"""Retain every DF shell launch/resource row and check its expected task grid.

The host model uses shell IDs from the bundled basis, never N^3 resident task
metadata. Nsight activity duration is separate from native stream-event timing.
"""
import argparse
import hashlib
import json
import math
import re
import sqlite3
from collections import defaultdict
from pathlib import Path

from vibeqc import Atom
from vibeqc.calculator import _named_basis_shells
from benchmarks._cases import benchmark_cases
from benchmarks.df_policy_endpoint import CASES


def expected(aos, grouped, schedules):
    case = benchmark_cases()[CASES[aos]]
    shells = _named_basis_shells(case.vibeqc_basis, [Atom.from_value(a) for a in case.atoms])
    offsets = [0]
    for s in shells:
        offsets.append(offsets[-1] + 2 * s.angular_momentum + 1)
    groups = defaultdict(list)
    for i, shell in enumerate(shells):
        groups[(shell.angular_momentum, len(shell.primitives) if grouped else 0)].append(i)
    groups = sorted(groups.items())
    start = 0
    while start < aos:
        end = min(aos, start + (52 if aos == 384 else 64))
        if aos == 768:
            end = max(x for x in offsets if x <= end)
        for ga, ((la, pa), sa) in enumerate(groups):
            for gb, ((lb, pb), sb) in enumerate(groups):
                if ga < gb:
                    continue
                pairs = len(sa)*(len(sa)+1)//2 if ga == gb else len(sa)*len(sb)
                for (lc, pc), sc in groups:
                    sc = [i for i in sc if offsets[i] < end and offsets[i+1] > start]
                    if not sc:
                        continue
                    tasks = pairs * len(sc)
                    lanes, block_groups = schedules[(la,lb,lc)]
                    yield {'angular': [la,lb,lc], 'nprim': [pa,pb,pc], 'panel': [start,end],
                           'tasks': tasks, 'grid_x': math.ceil(tasks/block_groups),
                           'block_x': lanes*block_groups}
        start = end


def packet_plan(aos, schedules):
    """Mirror bounded host packets; every signature remains an implicit product."""
    by_panel_class = defaultdict(list)
    for row in expected(aos, True, schedules):
        by_panel_class[(tuple(row['panel']), tuple(row['angular']))].append(row)
    for (_, _), slices in sorted(by_panel_class.items()):
        for start in range(0, len(slices), 24):
            ordered = sorted(slices[start:start+24], key=lambda s: -math.prod(s['nprim']))
            yield {'angular': ordered[0]['angular'], 'nprim': [],
                   'panel': ordered[0]['panel'], 'tasks': sum(s['tasks'] for s in ordered),
                   'grid_x': sum(s['grid_x'] for s in ordered),
                   'block_x': ordered[0]['block_x'], 'signatures': ordered}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('sqlite', type=Path)
    parser.add_argument('output', type=Path)
    parser.add_argument('--aos', type=int, required=True)
    parser.add_argument('--grouped', action='store_true')
    parser.add_argument('--packet', action='store_true')
    parser.add_argument('--generated', type=Path, default=Path('build/cuda-release-sm120/generated/generated_df_shell_derivatives.cuh'))
    args = parser.parse_args()
    generated = args.generated.read_text()
    schedules = {tuple(map(int,m[:3])):tuple(map(int,m[3:])) for m in re.findall(
        r'Schedule<(\d),(\d),(\d),2>\s*\{\s*static constexpr unsigned lanes=(\d+);\s*static constexpr unsigned groups=(\d+);',generated)}
    c = sqlite3.connect(args.sqlite)
    kernels = c.execute('''select s.value,k.start,k.end,k.gridX,k.blockX,k.registersPerThread,
        k.staticSharedMemory,k.dynamicSharedMemory,k.localMemoryPerThread,k.correlationId
        from CUPTI_ACTIVITY_KIND_KERNEL k join StringIds s on k.demangledName=s.id
        where s.value like '%::shell_panel<%' or s.value like '%::shell_packet<%' order by k.start''').fetchall()
    plan = list(packet_plan(args.aos,schedules) if args.packet else expected(args.aos,args.grouped,schedules))
    assert len(plan)==len(kernels), (len(plan),len(kernels))
    rows, summary = [], {}
    for model,kernel in zip(plan,kernels,strict=True):
        name,start,end,grid,block,registers,static,dynamic,local,correlation=kernel
        angular = list(map(int,re.findall(r'\(unsigned int\)(\d)',name)))[:3]
        assert angular == model['angular'], (angular,model)
        assert (grid,block)==(model['grid_x'],model['block_x']), (kernel,model)
        row={**model,'gpu_ms':(end-start)/1e6,'registers_per_thread':registers,
             'static_shared_bytes':static,'dynamic_shared_bytes':dynamic,
             'local_bytes_per_thread':local,'correlation_id':correlation}
        rows.append(row)
        key=tuple(angular+model['nprim'])
        total=summary.setdefault(key,{'angular':angular,'nprim':model['nprim'],'launches':0,'tasks':0,'gpu_ms':0.,'panel_ns_per_task':[]})
        total['launches']+=1
        total['tasks']+=model['tasks']
        total['gpu_ms']+=row['gpu_ms']
        total['panel_ns_per_task'].append((end-start)/model['tasks'])
    for row in summary.values():
        row['mean_ns_per_task']=row['gpu_ms']*1e6/row['tasks']
    payload={'aos':args.aos,'grouped':args.grouped or args.packet,'packet':args.packet,'sqlite_sha256':hashlib.sha256(args.sqlite.read_bytes()).hexdigest(),
             'generated_sha256':hashlib.sha256(generated.encode()).hexdigest(),
             'gpu_ms':sum(r['gpu_ms'] for r in rows),'launches':len(rows),
             'tasks':sum(r['tasks'] for r in rows),'all_launch_grids_verified':True,
             'signature_totals':list(summary.values()),'launch_records':rows,
             'interpretation':'Actual CUPTI kernel durations and launch resources. Task counts and signatures come from a bounded host reconstruction whose every launch class and grid matches CUPTI. Packet signatures share one measured duration and have no individual duration attribution. Panel ns/task includes wave/tail and task-content effects; it is not per-task clock sampling or achieved occupancy.'}
    c.row_factory = sqlite3.Row
    payload['captured_hardware'] = [dict(r) for r in c.execute('select name,computeMajor,computeMinor,totalMemory,smCount,maxRegistersPerSm,maxShmemPerSm,maxWarpsPerSm,maxBlocksPerSm from TARGET_INFO_GPU')]
    payload['captured_driver_metadata'] = [dict(r) for r in c.execute("select name,value from TARGET_INFO_SYSTEM_ENV where lower(name) like '%driver%' or lower(name) like '%cuda%version%'")]
    payload['complete_endpoint_memcpy'] = [{'copy_kind':kind, 'calls':calls, 'bytes':size, 'gpu_ms':duration/1e6} for kind,calls,size,duration in c.execute('select copyKind,count(*),sum(bytes),sum(end-start) from CUPTI_ACTIVITY_KIND_MEMCPY group by copyKind')]
    payload['memcpy_kind_enum'] = [dict(r) for r in c.execute('select id,label from ENUM_CUDA_MEMCPY_OPER')]
    payload['memcpy_scope'] = 'All memcpy activities in the captured measured warm energy+force endpoint, excluding prime/cold and implicit kernel parameters.'
    payload['complete_endpoint_kernel_launches'] = c.execute('select count(*) from CUPTI_ACTIVITY_KIND_KERNEL').fetchone()[0]
    payload['explicit_synchronization_api_calls'] = {name:count for name,count in c.execute("select s.value,count(*) from CUPTI_ACTIVITY_KIND_RUNTIME r join StringIds s on r.nameId=s.id where s.value like '%Synchronize%' group by s.value")}
    args.output.write_text(json.dumps(payload,indent=2)+'\n')
    print({k:payload[k] for k in ('aos','grouped','gpu_ms','launches','tasks')})


if __name__ == '__main__':
    main()
