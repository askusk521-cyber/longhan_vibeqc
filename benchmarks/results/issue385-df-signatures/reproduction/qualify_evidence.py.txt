"""Audit all publication inputs before creating the compact qualification record."""
import hashlib
import json
from pathlib import Path
import re
import statistics
import subprocess

root = Path('.artifacts/issue385')
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
measurements = {}
for stage, policies in [('baseline', ['auto']), ('baseline-refresh', ['auto']),
                        ('uniform', ['off', 'on']), ('packet', ['off', 'on', 'packet']),
                        ('final', ['off', 'auto'])]:
    measurements[stage] = {}
    sizes = (96, 192, 384, 768) if stage == 'final' else (384, 768)
    for aos in sizes:
        path = root / stage / f'{aos}-clean.json'
        data = json.loads(path.read_text())
        assert data['scope'] == 'clean endpoint'
        assert not data['controls'].get('VIBEQC_DF_TRACE')
        assert data['controls'].get('VIBEQC_DF_SHELL_COUNTERS', '0') == '0'
        by_policy = {}
        for policy in policies:
            samples = [s for s in data['samples'] if s['policy'] == policy]
            assert len(samples) == 5
            assert all(s['maximum_energy_error'] <= 1e-9 and s['maximum_force_error'] <= 1e-8 for s in samples)
            expected_iterations = 2 if aos == 96 else 3
            assert all(s['iterations'] == [expected_iterations] for s in samples)
            by_policy[policy] = {'samples': [s['seconds'] for s in samples],
                                 'median_seconds': statistics.median(s['seconds'] for s in samples),
                                 'maximum_energy_error': max(s['maximum_energy_error'] for s in samples),
                                 'maximum_force_error': max(s['maximum_force_error'] for s in samples)}
            by_policy[policy]['scf_iterations'] = expected_iterations
        measurements[stage][str(aos)] = by_policy
libraries = {'baseline': Path('/home/jzzeng/codes/vibeqc-issue382/.artifacts/issue385/baseline/libvibeqc.so')}
