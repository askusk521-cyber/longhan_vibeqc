"""Run DF through the unchanged Direct J/K comparison and acceptance runners.

The existing small DF matrix retains its original criteria. Supplemental
384/768 comparisons use the exact #383 scientific settings and stricter large
DF gates. Failures are recorded, never removed by branch filtering or retries.
"""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

assert os.environ.get('SLURM_JOB_ID')
root = Path('.artifacts/issue385')
target = root / 'direct-protocol'
target.mkdir(exist_ok=True)
assert not (target / 'run.json').exists()
baseline = Path('/home/jzzeng/codes/vibeqc-issue382/.artifacts/issue385/baseline/libvibeqc.so')
candidate = (root / 'final/libvibeqc.so').resolve()
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
payload = {'slurm_job_id': os.environ['SLURM_JOB_ID'],
           'protocol': 'Unmodified Direct J/K runner; ABBA interleaving, fixed engine-local post-cold dm0, one untimed priming replay, synchronized complete energy and force.',
           'runner_sha256': {p: sha(Path(p)) for p in ('benchmarks/compare_gpu4pyscf_batch.py', 'benchmarks/real_molecule_gate.py', 'benchmarks/_cases.py')},
           'libraries': {k: {'path': str(p), 'sha256': sha(p), 'source_patch_sha256': sha(p.with_name('source.patch'))} for k, p in [('baseline', baseline), ('candidate', candidate)]},
           'runs': []}


def run(label, command, library):
    env = os.environ.copy()
    env['VIBEQC_LIBRARY'] = str(library)
    env['VIBEQC_DF_PRIMITIVE_BUCKETS'] = 'auto'
    for key in ('VIBEQC_DF_TRACE', 'VIBEQC_DF_HOST_TRACE', 'VIBEQC_DF_PROGRESS_TRACE', 'VIBEQC_DF_SHELL_COUNTERS'):
        env.pop(key, None)
    print('Starting', label, flush=True)
    with (root / (label + '.log')).open('w') as log:
        result = subprocess.run(command, env=env, stdout=log, stderr=subprocess.STDOUT, check=False)
    payload['runs'].append({'label': label, 'command': command, 'library_sha256': sha(library), 'returncode': result.returncode})
    (target / 'run.json').write_text(json.dumps(payload, indent=2) + '\n')
    print('Finished', label, 'exit', result.returncode, flush=True)


for size in (96, 192):
    run(f'direct-protocol-matrix-{size}', [sys.executable, 'benchmarks/real_molecule_gate.py', '--size', str(size), '--density-fitting', 'cuda', '--repeats', '5', '--output-directory', str(target / f'matrix-{size}')], candidate)

for aos, case in [(384, 'water-hexadecamer-2s4-def2-svp-spherical'), (768, 'water-32mer-4s4-def2-svp-spherical')]:
    for variant, library in [('baseline', baseline), ('candidate', candidate)]:
        output = target / f'{aos}-{variant}.json'
        command = [sys.executable, 'benchmarks/compare_gpu4pyscf_batch.py', '--case', case, '--batch', '1', '--repeats', '5', '--density-fitting', 'cuda', '--max-iterations', '100', '--energy-tolerance', '1e-12', '--density-tolerance', '1e-10', '--reference-gradient-tolerance', '1e-10', '--screening-tolerance', '1e-12', '--maximum-energy-error', '1e-9', '--maximum-force-error', '1e-8', '--output', str(output), '--progress-output', str(root / f'direct-protocol-{aos}-{variant}.jsonl')]
        run(f'direct-protocol-{aos}-{variant}', command, library)
