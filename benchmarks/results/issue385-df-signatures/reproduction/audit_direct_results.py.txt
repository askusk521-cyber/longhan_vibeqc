"""Preserve failed external gates and separately check native baseline parity."""
import hashlib
import json
from pathlib import Path

root = Path('.artifacts/issue385')


def read(path):
    return json.loads(path.read_text())


def flatten(value):
    if isinstance(value, list):
        return [x for item in value for x in flatten(item)]
    return [value]


summary = read(root / 'direct/summary.json')
baseline_summary = read(root / 'direct-baseline/summary.json')
assert not summary['passed'] and not baseline_summary['passed']
assert all(p['passed'] == (p['ao_count'] == 192) for p in summary['points'])
assert all(not p['passed'] for p in baseline_summary['points'])
rows = []
for batch in (1, 4):
    filename = f'water-tetramer-def2-svp-spherical-b{batch}-none.json'
    paths = {'baseline': root / 'direct-baseline' / filename,
             'candidate': root / 'direct' / filename,
             'tighter_reference': root / 'direct-tighter-reference' / f'96-b{batch}.json'}
    records = {k: read(p) for k, p in paths.items()}
    baseline, candidate = records['baseline'], records['candidate']
    assert baseline['workload'] == candidate['workload']
    row = {'aos': 96, 'batch': batch, 'original_gate_passed': False,
           'original_baseline_gate': baseline['gate'],
           'original_candidate_gate': candidate['gate'],
           'tighter_reference_gate': records['tighter_reference']['gate'],
           'tighter_reference_interpretation': 'Diagnostic only. Neither batch passes the original force gate; batch 4 also has reference nonconvergence. No replacement certification.',
           'source_sha256': {k: hashlib.sha256(p.read_bytes()).hexdigest() for k, p in paths.items()}}
    for key in ('energies_hartree', 'forces_hartree_per_bohr'):
        diffs = []
        for left, right in zip(baseline['vibeqc']['warm_samples'], candidate['vibeqc']['warm_samples'], strict=True):
            a, b = flatten(left[key]), flatten(right[key])
            assert len(a) == len(b)
            diffs.append(max(abs(x-y) for x, y in zip(a, b, strict=True)))
        assert len(diffs) == 5 and max(diffs) <= 3e-11
        row['native_baseline_candidate_' + key + '_maximum_difference'] = max(diffs)
    for key, record in records.items():
        assert len(record['vibeqc']['warm_samples']) == 5
        row[key + '_native_median_seconds'] = record['vibeqc']['warm_median_seconds']
        row[key + '_accuracy_selection'] = record['accuracy']['gate_selection']
    rows.append(row)

payload = {'original_direct_matrix_passed': False,
           'direct_192_batch1_batch4_passed': True,
           'direct_96_failure_reproduces_on_exact_383': True,
           'native_96_baseline_candidate_parity_passed': True,
           'interpretation': 'DF scheduling does not introduce the existing Direct96 discrepancy. Original Direct gates remain failed and unchanged; this is not #206 completion evidence.',
           'cases': rows}
(root / 'direct-audit.json').write_text(json.dumps(payload, indent=2) + '\n')
print('Original Direct failures retained; all five paired native samples agree within original 3e-11 gates.')
