#!/usr/bin/env bash
set -euo pipefail
export PYTHONPATH=python:. OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1
export PATH="/group/software/cuda-12.9.1/bin:$PATH"
export VIBEQC_LIBRARY="$PWD/.artifacts/issue385/packet/libvibeqc.so"
export LD_LIBRARY_PATH="$PWD/.artifacts/issue385/packet:/group/software/cuda-12.9.1/lib64"
.artifacts/issue385/occupancy < .artifacts/issue385/occupancy-input.txt > .artifacts/issue385/baseline-uniform-occupancy.json
for aos in 384 768; do
  case "$aos" in
    384) case_name=water-hexadecamer-2s4-def2-svp-spherical ;;
    768) case_name=water-32mer-4s4-def2-svp-spherical ;;
  esac
  /tmp/vibeqc-pr-review-env/bin/python -m benchmarks.df_policy_endpoint --aos "$aos" --control VIBEQC_DF_PRIMITIVE_BUCKETS --policies off on packet --repeats 5 --reference "benchmarks/results/issue377-379-df/gpu4pyscf/$case_name.json" --output ".artifacts/issue385/packet/$aos-clean.json"
  nvidia-smi --query-compute-apps=pid,used_memory --format=csv,noheader,nounits -lms 100 > ".artifacts/issue385/packet/$aos-memory.csv" &
  sampler_pid=$!
  trap 'kill "$sampler_pid" 2>/dev/null || true' EXIT
  VIBEQC_DF_SHELL_COUNTERS=1 /tmp/vibeqc-pr-review-env/bin/python -m benchmarks.df_policy_endpoint --aos "$aos" --control VIBEQC_DF_PRIMITIVE_BUCKETS --policies off on packet --repeats 1 --trace --reference "benchmarks/results/issue377-379-df/gpu4pyscf/$case_name.json" --output ".artifacts/issue385/packet/$aos-work.json"
  kill "$sampler_pid"
  trap - EXIT
  nsys profile --trace=cuda,nvtx --sample=none --cpuctxsw=none --cuda-event-trace=false --capture-range=cudaProfilerApi --capture-range-end=stop --output ".artifacts/issue385/packet/$aos-nsys" /tmp/vibeqc-pr-review-env/bin/python .artifacts/issue385/profile_endpoint.py --aos "$aos" --control VIBEQC_DF_PRIMITIVE_BUCKETS --policies packet --repeats 1 --cuda-profile --reference "benchmarks/results/issue377-379-df/gpu4pyscf/$case_name.json" --output ".artifacts/issue385/packet/$aos-nsys.json"
done

export LD_LIBRARY_PATH=/tmp/vibeqc-308-gpu4pyscf-env/lib/python3.13/site-packages/cutensor/lib:/group/software/cuda-12.9.1/lib64
for aos in 384 768; do
  /tmp/vibeqc-308-gpu4pyscf-env/bin/python .artifacts/issue385/reference/gpu4_reference.py "$aos" ".artifacts/issue385/reference/$aos-clean.json"
done
