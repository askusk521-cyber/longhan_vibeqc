"""Summarize clean endpoints and separately scoped diagnostics without resampling."""
from collections import Counter
import argparse
import hashlib
import json
import math
import re
import statistics
from pathlib import Path

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('directory', type=Path)
parser.add_argument('output', type=Path)
args = parser.parse_args()
rows = []
for aos in (96, 192, 384, 768):
    clean = args.directory / f'{aos}-clean.json'
    if not clean.exists():
        continue
    data = json.loads(clean.read_text())
    work = args.directory / f'{aos}-work.json'
    diagnostics = json.loads(work.read_text())['samples'] if work.exists() else []
    for policy in data['policies']:
        samples = [s for s in data['samples'] if s['policy'] == policy]
        if not samples:
            continue
        row = {'aos': aos, 'policy': policy, 'clean_samples': [s['seconds'] for s in samples],
               'median_seconds': statistics.median(s['seconds'] for s in samples),
               'maximum_energy_error': max(s['maximum_energy_error'] for s in samples),
               'maximum_force_error': max(s['maximum_force_error'] for s in samples),
               'iterations': [s['iterations'] for s in samples],
               'library_sha256': data['library_sha256'],
               'source_identity': data['native_source_identity'],
               'source_patch_sha256': data['source_patch_sha256'],
               'clean_sha256': hashlib.sha256(clean.read_bytes()).hexdigest()}
        diagnostic = next((s for s in diagnostics if s['policy'] == policy), None)
        if diagnostic:
            group = next(g for g in diagnostic['components']['groups'] if g['operation'] == 'force_response')
            gpu, counters = group['gpu_exclusive_ms'], group['counter_sums']
            signatures = {k: v for k, v in counters.items() if re.fullmatch(r'shell_\d\d\d_p\d+_\d+_\d+', k)}
            if signatures:
                assert sum(signatures.values()) == counters['shell_triples_visited']
                products = sum(math.prod(map(int, k.split('_p')[1].split('_'))) * v for k, v in signatures.items())
                if products:
                    assert products == counters['shell_primitive_products']
            raw_trace = work.with_suffix(f'.{diagnostic["repeat"]}-{policy}.jsonl')
            launch_counts = Counter()
            if raw_trace.exists():
                for line in raw_trace.read_text().splitlines():
                    record = json.loads(line)
                    if record.get('operation') == 'force_response':
                        for region in record['regions']:
                            match = re.fullmatch(r'shell_(\d\d\d)_(p\d+_\d+_\d+|packet)', region['name'])
                            if match:
                                launch_counts[match[1]] += 1
            resources = {}
            for name, value in group['counter_maxima'].items():
                match = re.fullmatch(r'shell_(\d\d\d)_(registers|static_shared_bytes|dynamic_shared_bytes|resident_block_limit|resident_thread_limit|sm_thread_limit)', name)
                if match:
                    divisor = 1 if counters.get('shell_resource_values_are_maxima') else launch_counts[match[1]]
                    assert divisor > 0 and value % divisor == 0
                    resources[name] = value // divisor
            row['diagnostic'] = {
                'force_stage_seconds': diagnostic['force_stage_seconds'],
                'three_center_event_interval_ms': gpu.get('three_center_derivative_contraction', 0)
                    + sum(v for k, v in gpu.items() if re.fullmatch(r'shell_\d\d\d_(p\d+_\d+_\d+|packet)', k)),
                'exclusive_gpu_ms': gpu, 'counters': counters,
                'normalized_launch_resources': resources,
                'class_launch_counts': dict(launch_counts),
                'resource_note': 'Uniform prototype raw resources sum repeated launches; divide by observed trace launch count. A marker identifies final per-operation maxima. Raw counters are retained unchanged.',
                'resident_process_bytes': diagnostic.get('process_device_resident_bytes')}
            host = work.with_suffix(f'.{diagnostic["repeat"]}-{policy}.host.jsonl')
            if host.exists():
                regions = [r for line in host.read_text().splitlines()
                           for r in json.loads(line).get('regions', [])]
                row['diagnostic']['host_metadata_ms'] = sum(r['wall_ms'] for r in regions if r['name'] == 'df_shell_metadata')
                row['diagnostic']['host_dispatch_ms_including_backpressure'] = sum(r['wall_ms'] for r in regions if r['name'] == 'df_shell_signature_dispatch')
        memory = args.directory / f'{aos}-memory.csv'
        if memory.exists():
            mib = [int(line.split(',')[1]) for line in memory.read_text().splitlines() if ',' in line]
            row['all_policy_diagnostic_sampled_peak_bytes'] = max(mib, default=0) * 1024**2
            row['peak_scope'] = '100 ms process sampler over this workload diagnostic run across all policies; a sampled lower bound, not an exact allocator high-water mark.'
        rows.append(row)
args.output.write_text(json.dumps(rows, indent=2) + '\n')
for r in rows:
    d = r.get('diagnostic', {})
    print(r['aos'], r['policy'], r['median_seconds'], d.get('force_stage_seconds'), d.get('three_center_event_interval_ms'))
